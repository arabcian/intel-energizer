cmake_minimum_required(VERSION 3.20)
project(intel-energizer-gui VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
add_library(energizer-panels STATIC inteltab.cpp cpustressbox.cpp cpustress.cpp
    memtestbox.cpp memtest.cpp charts.cpp theme.cpp trialbar.cpp privileged.cpp topology.cpp diagreport.cpp)
target_compile_definitions(energizer-panels PUBLIC
    CENTURION_HELPER_DIR="${CMAKE_INSTALL_FULL_LIBEXECDIR}/intel-energizer"
    CENTURION_VERSION="${PROJECT_VERSION}")
target_link_libraries(energizer-panels PUBLIC Qt6::Widgets)
add_executable(intel-energizer-gui main.cpp theme.qrc)
target_link_libraries(intel-energizer-gui PRIVATE energizer-panels)
find_package(Qt6 QUIET COMPONENTS LinguistTools)
if(TARGET Qt6::lrelease)
    qt_add_translations(intel-energizer-gui TS_FILES intel-energizer_tr.ts RESOURCE_PREFIX "/i18n")
endif()
install(TARGETS intel-energizer-gui RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
include(CTest)
if(BUILD_TESTING)
    add_executable(window-test tests/window_test.cpp)
    target_link_libraries(window-test PRIVATE energizer-panels)
    add_test(NAME window COMMAND window-test)
    set_tests_properties(window PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;CENTURION_PGO_TRAIN=1")
    foreach(panel cpustress memtest)
        add_executable(${panel}-test tests/${panel}_test.cpp ${panel}.cpp)
        target_link_libraries(${panel}-test PRIVATE Qt6::Widgets)
        add_test(NAME ${panel} COMMAND ${panel}-test)
    endforeach()
endif()
install(FILES intel-energizer.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
configure_file(../packaging/intel-energizer-daemon.service.in intel-energizer-daemon.service @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/intel-energizer-daemon.service DESTINATION ${CMAKE_INSTALL_LIBDIR}/systemd/system)
